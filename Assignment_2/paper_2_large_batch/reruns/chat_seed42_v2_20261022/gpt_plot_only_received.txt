def fresh_workflow(source):
    # Hash and execute the same source bytes in a new namespace for every analysis.
    fingerprint = hashlib.sha256(source.encode("utf-8")).hexdigest()
    module = types.ModuleType("figure3_saved_results")
    module.__dict__["SOURCE_SNAPSHOT"] = source
    exec(compile(source, f"<figure3-analysis-{fingerprint}>", "exec"), module.__dict__)
    return module
